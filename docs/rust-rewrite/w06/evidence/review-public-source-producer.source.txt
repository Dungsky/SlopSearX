def require(value, message):
    if not value:
        raise RuntimeError(message)


def safe(value):
    if dataclasses.is_dataclass(value):
        return safe(dataclasses.asdict(value))
    if isinstance(value, dict):
        return {key: safe(item) for key, item in value.items()}
    if isinstance(value, (tuple, list)):
        return [safe(item) for item in value]
    if isinstance(value, set):
        return sorted(safe(item) for item in value)
    return value

def instrument(state, trace, patch):
    original_resolve = composition.resolve_source

    async def resolved(source, destination):
        result = await original_resolve(source, destination)
        trace.append(
            {
                "method": "resolve_source",
                "source": safe(source),
                "destination": destination,
                "result": safe(result.stored() if isinstance(result, composition.ResolvedSource) else result),
            }
        )
        return result

    patch.setattr(staged_tools, "resolve_source", resolved)
    for name in ["find_by_idempotency", "admit", "read", "request_retry", "save_owned"]:
        original = getattr(state.staged_store, name)

        async def wrapped(*pos, _name=name, _original=original, **kw):
            trace.append({"method": _name, "phase": "called", "arguments": safe(pos)})
            result = await _original(*pos, **kw)
            trace.append({"method": _name, "phase": "returned", "result": safe(result)})
            return result

        patch.setattr(state.staged_store, name, wrapped)


async def outcome(awaitable):
    try:
        result = safe(await awaitable)
        captured = {"output": result}
        if isinstance(result, dict) and isinstance(result.get("operation_id"), str):
            state = staged_tools.get_state()
            key = state.staged_store._key("default", result["operation_id"])
            captured["after_record"] = copy.deepcopy(state.staged_store._store._data.get(key))
        return captured
    except Exception as exc:
        return {"exception": {"class": type(exc).__name__, "message": str(exc)}}
