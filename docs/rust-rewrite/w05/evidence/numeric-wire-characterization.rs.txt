
#[tokio::test]
async fn actual_historical_numeric_wire_characterization() {
    for granted in [true] {
        let server = w05_support::Server::start();
        let engine = Arc::new(FixtureEngine(0.into()));
        let deps = search_support::dependencies(vec![("brave", engine.clone())]);
        let policy = deps.policy.clone();
        let search = SearchService::new(SearchConfig::default(), deps).unwrap();
        let store = SavedStore::new(
            Arc::new(server.store()),
            TenantId::new("tenant:wire").unwrap(),
        );
        let authority = Arc::new(StaticSavedAuthority {
            search: policy.clone(),
            saved_granted: true,
            auth_configured: BTreeMap::from([("brave".into(), false)]),
            limits: SavedLimits::default(),
        });
        let publisher = EventPublisher {
            limits: EventLimits::default(),
            authority: Arc::new(StaticEventAuthority {
                search: policy,
                events_granted: granted,
                saved_granted: true,
            }),
            observer: Arc::new(NoopEventObserver),
        };
        let principal =
            EventPrincipal::new(store.tenant().clone(), "trusted-fixture-subject".into()).unwrap();
        let events = EventService::new(store.clone(), &principal, publisher);
        let registry =
            SavedEventRegistry::new(events, search.clone(), authority, Duration::from_secs(2))
                .unwrap();
        let cancel = CancellationToken::new();
        let router = http_router(registry.clone(), cancel.clone());
        let listener = tokio::net::TcpListener::bind("127.0.0.1:0").await.unwrap();
        let address = listener.local_addr().unwrap();
        assert!(address.ip().is_loopback());
        let url = format!("http://{address}/mcp");
        let parsed = url::Url::parse(&url).unwrap();
        assert_eq!(parsed.scheme(), "http");
        assert_eq!(parsed.host_str(), Some("127.0.0.1"));
        assert_eq!(parsed.port(), Some(address.port()));
        let shutdown = cancel.clone();
        let host = tokio::spawn(async move {
            axum::serve(listener, router)
                .with_graceful_shutdown(shutdown.cancelled_owned())
                .await
                .unwrap();
        });
        // no_proxy prevents inherited proxy settings from routing this fixture.
        let http = reqwest::Client::builder()
            .no_proxy()
            .redirect(reqwest::redirect::Policy::none())
            .build()
            .unwrap();

        let mut connection = redis::Client::open(server.url.as_str()).unwrap().get_connection().unwrap();
        let now = store.now().await.unwrap();
        let raw_payload = format!(r#"{{"event_id":"historical-number","event_type":"report_created","occurred_at":{now},"_policy_engines":["brave"],"first":18446744073709551617,"second":18446744073709551618,"reserved":{{"$serde_json::private::Number":"123"}}}}"#);
        let event=ssx_infra::saved_events::codec::Event::from_payload(ssx_json::from_str(&raw_payload).unwrap()).unwrap();
        let stream=ssx_infra::valkey::keys::saved("events",store.tenant(),None).unwrap();
        let _:String=redis::cmd("XADD").arg(stream).arg("*").arg("payload").arg(event.bytes().unwrap()).query(&mut connection).unwrap();
        let init=http.post(&url).header("accept","application/json, text/event-stream").json(&json!({"jsonrpc":"2.0","id":1,"method":"initialize","params":{"protocolVersion":"2025-11-25","capabilities":{},"clientInfo":{"name":"w05-raw-number","version":"1"}}})).send().await.unwrap();
        assert!(init.status().is_success());
        let session=init.headers().get("mcp-session-id").cloned();
        let _=init.bytes().await.unwrap();
        let mut ready=http.post(&url).header("accept","application/json, text/event-stream").header("mcp-protocol-version","2025-11-25").json(&json!({"jsonrpc":"2.0","method":"notifications/initialized"}));
        if let Some(s)=&session {ready=ready.header("mcp-session-id",s);}
        assert!(ready.send().await.unwrap().status().is_success());
        let mut request=http.post(&url).header("accept","application/json, text/event-stream").header("mcp-protocol-version","2025-11-25").json(&json!({"jsonrpc":"2.0","id":2,"method":"tools/call","params":{"name":"slopsearx_read_saved_search_events","arguments":{"consumer_id":"raw-numbers"}}}));
        if let Some(s)=&session {request=request.header("mcp-session-id",s);}
        let response=request.send().await.unwrap();assert!(response.status().is_success());
        let bytes=response.bytes().await.unwrap();
        let wire:ssx_json::Value=ssx_json::from_slice(&bytes).unwrap();
        let text:ssx_json::Value=ssx_json::from_str(wire["result"]["content"][0]["text"].as_str().unwrap()).unwrap();
        let structured=&wire["result"]["structuredContent"];
        assert_eq!(text["returned"],ssx_json::json!(1));
        assert_ne!(text["events"][0]["first"],text["events"][0]["second"]);
        assert_eq!(text["events"][0]["reserved"],ssx_json::json!({"$serde_json::private::Number":"123"}));
        println!("W05_NUMERIC_RAW_WIRE:{}",String::from_utf8(bytes.to_vec()).unwrap());
        assert_ne!(&text,structured,"characterization requires SDK loss to be observed");
        assert_eq!(structured["events"][0]["first"],structured["events"][0]["second"]);
        registry.close().await; search.shutdown().await; cancel.cancel();
        tokio::time::timeout(Duration::from_secs(3),host).await.unwrap().unwrap();
        drop(connection);drop(registry);drop(store);drop(server);
    }
}
