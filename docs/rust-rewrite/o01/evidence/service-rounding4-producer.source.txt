class Pipeline:
    def __init__(self):
        self.commands = []
        self.executed = asyncio.Event()
    def __getattr__(self, name):
        def command(*args):
            self.commands.append([name, *args])
            return self
        return command
    async def execute(self):
        self.executed.set()
class Cache:
