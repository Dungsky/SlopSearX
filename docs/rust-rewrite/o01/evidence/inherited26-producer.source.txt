class Pipeline:
    def __init__(self, fail=False):
        self.calls = []
        self.fail = fail

    def __getattr__(self, name):
        def command(*args):
            self.calls.append([name, *args])
            return self

        return command

    async def execute(self):
        if self.fail:
            raise RuntimeError("secret failure must not escape")


class Cache:
    is_connected = True

    def __init__(self, fail=False):
        self._client = self
        self.pipe = Pipeline(fail)

    def pipeline(self):
        return self.pipe
