def np5(selected,g):
 assert len(selected)==5 and len(set(selected))==5
 ideal=sum(sorted(g.values(),reverse=True)[:5])

class Lexical:
 def __init__(self,resources):
  self.count=len(resources);self.lens={};self.post=collections.defaultdict(dict)
  for rid,r in sorted(resources.items()):
   ts=tok(r['name']+' '+r['URL']+' '+r['Vertical']);self.lens[rid]=len(ts)
   for t,n in collections.Counter(ts).items():self.post[t][rid]=n
  self.avg=sum(self.lens.values())/self.count
 def rank(self,q):
  scores=collections.defaultdict(float)
  for t in sorted(set(tok(q))):
   post=self.post.get(t,{})
   if not post:continue
   idf=math.log(1+(self.count-len(post)+.5)/(len(post)+.5))
   for rid,n in post.items():scores[rid]+=idf*n*2.2/(n+1.2*(.25+.75*self.lens[rid]/self.avg))
  return sorted(self.lens,key=lambda rid:(-scores[rid],rid))
