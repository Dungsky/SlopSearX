class BM25:
 def __init__(self,corpus,title_only):
  self.ids=sorted(corpus);self.count=len(self.ids);self.length={};self.post=collections.defaultdict(dict)
  for did in self.ids:
   d=corpus[did];ts=tok(d['title']+('' if title_only else ' '+d['text']));self.length[did]=len(ts)
   for t,f in collections.Counter(ts).items():self.post[t][did]=f
  self.avg=sum(self.length.values())/self.count
 def search(self,q):
  score=collections.defaultdict(float)
  for t in sorted(set(tok(q))):
   post=self.post.get(t,{})
   if not post:continue
   idf=math.log(1+(self.count-len(post)+.5)/(len(post)+.5))
   for did,f in post.items():score[did]+=idf*f*2.2/(f+1.2*(.25+.75*self.length[did]/self.avg))
  # Always retain top20; explicit zero-score padding is benchmark fixture behavior.
  return [(d,score[d]) for d in sorted(self.ids,key=lambda d:(-score[d],d))[:20]]

class RankK:
 def __init__(self,k):self.k=k
 def rank(self,feeds,q,params=None):
  seen={}
  for en in sorted(feeds):
   contributed=set()
   for rank,r in enumerate(feeds[en],1):
    url=_normalise_url(r.url)
    if url in contributed:continue
    contributed.add(url);v=1/(self.k+rank)
    if url in seen:seen[url].score+=v;seen[url].engines.add(en);seen[url].tier=min(seen[url].tier,r.tier)
    else:seen[url]=dataclasses.replace(r,engine=en,engines={en},score=v)
  out=sorted(seen.items(),key=lambda x:(x[1].tier,-x[1].score,x[0]))
  for i,(_,r) in enumerate(out,1):r.position=i
  return [r for _,r in out]

def mmr(original):
 left=list(enumerate(original));selected=[];maximum=max((r.score for r in original),default=1)
 sets={r.url:set(tok(r.title+' '+r.content)) for r in original}
 while left:
  tier=min(r.tier for _,r in left)
  def utility(item):
   i,r=item;a=sets[r.url]
   redundancy=max((len(a&sets[t.url])/max(1,len(a|sets[t.url])) for t in selected),default=0)
   return .85*r.score/maximum-.15*redundancy
  i,r=max((t for t in left if t[1].tier==tier),key=lambda t:(utility(t),-t[0]));selected.append(r);left.remove((i,r))
 return selected

def ndcg(ids,grades,k=10):
 gain=lambda values:sum((2**v-1)/math.log2(i+2) for i,v in enumerate(values[:k]))
 ideal=gain(sorted(grades.values(),reverse=True));return gain([grades.get(d,0) for d in ids])/ideal if ideal else 0

def interval(values):

def recall(ids,grades,k=10):
 relevant={d for d,v in grades.items() if v>0};return len(set(ids[:k])&relevant)/len(relevant) if relevant else 0
